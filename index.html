    function loadIntoCropper(event) {

        const file = event.target.files[0];

        if (!file) return;


        const reader = new FileReader();

        reader.onload = function(e) {

    // 3. Adaptive Preprocessing + Clean Digit Extraction

    async function confirmCropAndScan() {

        if (!cropper) return;


        // Get cropped square at crisp 900x900 resolution

        const croppedCanvas = cropper.getCroppedCanvas({ width: 900, height: 900 });

        cropSection.style.display = 'none';

        clearBoard();

        statusElement.innerText = "⏳ Enhancing image & removing grid lines...";


        const ctx = croppedCanvas.getContext('2d');

        const size = 900;

        const imgData = ctx.getImageData(0, 0, size, size);

        const gray = new Uint8Array(size * size);


        // Convert to grayscale

        for (let i = 0; i < size * size; i++) {

            const p = i * 4;

            gray[i] = 0.299 * imgData.data[p] + 0.587 * imgData.data[p + 1] + 0.114 * imgData.data[p + 2];

        }


        // Integral image for fast Adaptive Mean Thresholding (handles phone shadows!)

        const integral = new Uint32Array((size + 1) * (size + 1));

        for (let y = 0; y < size; y++) {

            let rowSum = 0;

            for (let x = 0; x < size; x++) {

                rowSum += gray[y * size + x];

                integral[(y + 1) * (size + 1) + (x + 1)] = integral[y * (size + 1) + (x + 1)] + rowSum;

            }

        }


        const binary = new Uint8Array(size * size); // 1 = ink (black), 0 = background (white)

        const radius = 25; // Local window size

        const C = 18;      // Contrast sensitivity


        for (let y = 0; y < size; y++) {

            const y1 = Math.max(0, y - radius);

            const y2 = Math.min(size - 1, y + radius);

            for (let x = 0; x < size; x++) {

                const x1 = Math.max(0, x - radius);

                const x2 = Math.min(size - 1, x + radius);

                const count = (x2 - x1 + 1) * (y2 - y1 + 1);

                const sum = integral[(y2 + 1) * (size + 1) + (x2 + 1)]

                          - integral[y1 * (size + 1) + (x2 + 1)]

                          - integral[(y2 + 1) * (size + 1) + x1]

                          + integral[y1 * (size + 1) + x1];

                const mean = sum / count;

                binary[y * size + x] = gray[y * size + x] < (mean - C) ? 1 : 0;

            }

        }


        try {

            const worker = await Tesseract.createWorker('eng');

            await worker.setParameters({

                tessedit_char_whitelist: '123456789',

                tessedit_pageseg_mode: Tesseract.PSM.SINGLE_CHAR,

            });


            const cellDim = 100; // 900 / 9

            const cleanCanvas = document.createElement('canvas');

            cleanCanvas.width = 120;

            cleanCanvas.height = 120;

            const cleanCtx = cleanCanvas.getContext('2d');


            for (let r = 0; r < 9; r++) {

                statusElement.innerText = `🔍 Reading row ${r + 1} of 9...`;

                for (let c = 0; c < 9; c++) {

                    const isolatedDigit = extractCleanDigitFromCell(binary, size, c * cellDim, r * cellDim, cellDim);

                    if (!isolatedDigit) continue; // Empty cell


                    // Render isolated digit centered on a clean white 120x120 canvas

                    cleanCtx.fillStyle = '#FFFFFF';

                    cleanCtx.fillRect(0, 0, 120, 120);

                    const { minX, minY, w, h, cellBin } = isolatedDigit;

                    const offsetX = Math.floor((120 - w) / 2);

                    const offsetY = Math.floor((120 - h) / 2);


                    cleanCtx.fillStyle = '#000000';

                    for (let dy = 0; dy < h; dy++) {

                        for (let dx = 0; dx < w; dx++) {

                            if (cellBin[(minY + dy) * cellDim + (minX + dx)] === 1) {

                                cleanCtx.fillRect(offsetX + dx, offsetY + dy, 1, 1);

                            }

                        }

                    }


                    const { data: { text } } = await worker.recognize(cleanCanvas);

                    const digit = text.trim().replace(/[^1-9]/g, '');

                    if (digit.length === 1) {

                        cells[r * 9 + c].value = digit;

                    }

                }

            }


            await worker.terminate();

            statusElement.innerText = "✅ Scan finished! Fix any wrong digit if needed, then click Solve.";

        } catch (err) {

            console.error(err);

            statusElement.innerText = "❌ OCR Error. Please check your internet connection and retry.";

        }

    }


    // Flood-fills away border grid lines and isolates the central digit bounding box

    function extractCleanDigitFromCell(globalBin, globalSize, startX, startY, cellDim) {

        const cellBin = new Uint8Array(cellDim * cellDim);

        for (let y = 0; y < cellDim; y++) {

            for (let x = 0; x < cellDim; x++) {

                cellBin[y * cellDim + x] = globalBin[(startY + y) * globalSize + (startX + x)];

            }

        }


        // 1. Erase any black pixels connected to the outer 12px border (removes grid lines & specks)

        const margin = 12;

        const queue = [];

        for (let y = 0; y < cellDim; y++) {

            for (let x = 0; x < cellDim; x++) {

                if ((x < margin || x >= cellDim - margin || y < margin || y >= cellDim - margin) && cellBin[y * cellDim + x] === 1) {

                    cellBin[y * cellDim + x] = 0;

                    queue.push(y * cellDim + x);

                }

            }

        }


        while (queue.length > 0) {

            const idx = queue.pop();

            const cx = idx % cellDim;

            const cy = Math.floor(idx / cellDim);

            const neighbors = [

                [cx - 1, cy], [cx + 1, cy], [cx, cy - 1], [cx, cy + 1]

            ];

            for (const [nx, ny] of neighbors) {

                if (nx >= 0 && nx < cellDim && ny >= 0 && ny < cellDim) {

                    const nIdx = ny * cellDim + nx;

                    if (cellBin[nIdx] === 1) {

                        cellBin[nIdx] = 0;

                        queue.push(nIdx);

                    }

                }

            }

        }


        // 2. Find bounding box of the remaining central digit

        let minX = cellDim, maxX = 0, minY = cellDim, maxY = 0, inkCount = 0;

        for (let y = margin; y < cellDim - margin; y++) {

            for (let x = margin; x < cellDim - margin; x++) {

                if (cellBin[y * cellDim + x] === 1) {

                    inkCount++;

                    if (x < minX) minX = x;

                    if (x > maxX) maxX = x;

                    if (y < minY) minY = y;

                    if (y > maxY) maxY = y;

                }

            }

        }


        // Ignore noise specks (too few pixels or tiny height)

        const w = maxX - minX + 1;

        const h = maxY - minY + 1;

        if (inkCount < 45 || h < 18 || w < 6) return null;


        return { minX, minY, w, h, cellBin };

    }


    // 4. Backtracking Sudoku Solver

    function handleSolve() {

        const grid = [];

        for (let r = 0; r < 9; r++) {

            const row = [];

            for (let c = 0; c < 9; c++) {

                row.push(parseInt(cells[r * 9 + c].value) || 0);

            }

            grid.push(row);

        }


        if (!isValidInitialBoard(grid)) {

            statusElement.innerText = "⚠️ Invalid puzzle! Duplicate numbers exist in a row, column, or 3×3 box.";

            return;

        }


        if (solveSudoku(grid)) {

            for (let r = 0; r < 9; r++) {

                for (let c = 0; c < 9; c++) {

                    const index = r * 9 + c;

                    if (cells[index].value === '') {

                        cells[index].value = grid[r][c];

                        cells[index].classList.add('solved');

                    }

                }

            }

            statusElement.innerText = "🎉 Sudoku Solved!";

        } else {

            statusElement.innerText = "❌ No valid solution found. Check if a scanned digit was misread.";

        }

    }


    function isValidInitialBoard(board) {

        for (let r = 0; r < 9; r++) {

            for (let c = 0; c < 9; c++) {

                const num = board[r][c];

                if (num !== 0) {

                    board[r][c] = 0;

                    if (!isSafe(board, r, c, num)) {

                        board[r][c] = num;

                        return false;

                    }

                    board[r][c] = num;

                }

            }

        }

        return true;

    }


    function isSafe(board, row, col, num) {

        for (let x = 0; x < 9; x++) {

            if (board[row][x] === num || board[x][col] === num) return false;

        }

        const startRow = row - (row % 3);

        const startCol = col - (col % 3);

        for (let i = 0; i < 3; i++) {

            for (let j = 0; j < 3; j++) {

                if (board[i + startRow][j + startCol] === num) return false;

            }

        }

        return true;

    }


    function solveSudoku(board) {

        for (let row = 0; row < 9; row++) {

            for (let col = 0; col < 9; col++) {

                if (board[row][col] === 0) {

                    for (let num = 1; num <= 9; num++) {

                        if (isSafe(board, row, col, num)) {

                            board[row][col] = num;

                            if (solveSudoku(board)) return true;

                            board[row][col] = 0;

                        }

                    }

                    return false;

                }

            }

        }

        return true;

    }
